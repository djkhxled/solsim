# SolSim v1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A Chrome MV3 extension that paper-trades Solana memecoins on gmgn.ai, reading GMGN's own live feed, and makes PnL share cards.

**Architecture:** A MAIN-world hook copies GMGN WebSocket/HTTP data and forwards it to an isolated-world panel. Pure ES-module logic (parse, price state, fills, positions, limits, ticks, ledger, card model) is unit-tested in Node. A module service worker is the single writer to `chrome.storage.local`. The popup and Card Studio are extension pages.

**Tech Stack:** Plain JavaScript ES modules, Chrome Manifest V3, Canvas 2D, `node --test` (Node 26). No dependencies, no build step.

**Spec:** `docs/superpowers/specs/2026-11-05-solsim-design.md`

## Global Constraints

- No wallet permissions, no signing code, no private keys, no real trades.
- No remote code. Every script ships in the folder.
- Network hosts: gmgn.ai (+ subdomains), lite-api.jup.ag, api.dexscreener.com, api.geckoterminal.com only.
- All state lives in `chrome.storage.local`.
- No GMGN logo, artwork or invite code on cards.
- Defaults: startingBalance 10, platformFeePct 1, priorityFeeSol 0.005, slippagePct 1, buyPresets [0.1,0.2,0.5,1,3,5,10,20], sellPresets [10,25,50,100], pnlCurrency "USD".
- Card sizes: landscape 1600×900, portrait 1080×1350.
- Live→backup after 3 s without a live tick. Stale after 5 s without any price. Backup poll every 2 s. Fetch timeout 4 s.
- Console log prefix: `[SolSim]`.
- Tests: `npm test` → `node --test "tests/**/*.test.js"`.

```js
export function fmtCompactUsd(n) {
  if (n == null || !Number.isFinite(n)) return '—';
  const sign = n < 0 ? '-' : '';
  const a = Math.abs(n);
  if (a >= 1e9) return `${sign}$${(a / 1e9).toFixed(2)}B`;
  if (a >= 1e6) return `${sign}$${(a / 1e6).toFixed(2)}M`;
  if (a >= 1e3) return `${sign}$${(a / 1e3).toFixed(2)}K`;
  return `${sign}$${a.toFixed(2)}`;
}

export function fmtUsd(n) {
  if (n == null || !Number.isFinite(n)) return '—';
  const a = Math.abs(n);
  if (a >= 1e3) return fmtCompactUsd(n);
  if (a === 0) return '$0';
  const sign = n < 0 ? '-' : '';
  return a >= 1 ? `${sign}$${a.toFixed(2)}` : `${sign}$${a.toPrecision(3)}`;
}

export function fmtSol(n, digits = 3) {
  if (n == null || !Number.isFinite(n)) return '—';
  const v = Number(n.toFixed(Math.abs(n) >= 100 ? 1 : digits));
  return String(v === 0 ? 0 : v);
}

export function fmtPct(n) {
  if (n == null || !Number.isFinite(n)) return '—';
  return `${n > 0 ? '+' : ''}${n.toFixed(Math.abs(n) >= 100 ? 0 : 2)}%`;
}

export function parseCompact(s) {
  const m = /^\s*\$?\s*([0-9]*\.?[0-9]+)\s*([kmb])?\s*$/i.exec(String(s ?? ''));
  if (!m) return null;
  const mult = { k: 1e3, m: 1e6, b: 1e9 }[(m[2] || '').toLowerCase()] || 1;
  return Number(m[1]) * mult;
}
```

```js
const CRC_TABLE = Array.from({ length: 256 }, (_, n) => {
  let c = n;
  for (let k = 0; k < 8; k++) c = c & 1 ? 0xedb88320 ^ (c >>> 1) : c >>> 1;
  return c >>> 0;
});
function crc32(buf) {
  let c = 0xffffffff;
  for (const b of buf) c = CRC_TABLE[(c ^ b) & 0xff] ^ (c >>> 8);
  return (c ^ 0xffffffff) >>> 0;
}
function chunk(type, data) {
  const len = Buffer.alloc(4); len.writeUInt32BE(data.length);
  const td = Buffer.concat([Buffer.from(type), data]);
  const crc = Buffer.alloc(4); crc.writeUInt32BE(crc32(td));
  return Buffer.concat([len, td, crc]);
}
export function encodePng(w, h, rgba) {
  const stride = w * 4 + 1;
  const raw = Buffer.alloc(stride * h);
  for (let y = 0; y < h; y++) rgba.copy(raw, y * stride + 1, y * w * 4, (y + 1) * w * 4);
  const ihdr = Buffer.alloc(13);
  ihdr.writeUInt32BE(w, 0); ihdr.writeUInt32BE(h, 4); ihdr[8] = 8; ihdr[9] = 6;
  return Buffer.concat([Buffer.from([137, 80, 78, 71, 13, 10, 26, 10]), chunk('IHDR', ihdr), chunk('IDAT', deflateSync(raw)), chunk('IEND', Buffer.alloc(0))]);
}
function segDist(px, py, [ax, ay], [bx, by]) {
  const dx = bx - ax, dy = by - ay;
  const t = Math.max(0, Math.min(1, ((px - ax) * dx + (py - ay) * dy) / (dx * dx + dy * dy)));
  return Math.hypot(px - (ax + t * dx), py - (ay + t * dy));
}
function sample(u, v) {
  const r = 0.22;
  const qx = Math.max(Math.abs(u - 0.5) - (0.5 - r), 0), qy = Math.max(Math.abs(v - 0.5) - (0.5 - r), 0);
  if (Math.hypot(qx, qy) > r) return [0, 0, 0, 0];
  let d = Infinity;
  for (let i = 0; i < LOGO_LINE.length - 1; i++) d = Math.min(d, segDist(u, v, LOGO_LINE[i], LOGO_LINE[i + 1]));
  if (d <= 0.045) return [255, 255, 255, 255];
  const t = (u + v) / 2;
  return [Math.round(0x14 + (0x99 - 0x14) * t), Math.round(0xf1 + (0x45 - 0xf1) * t), Math.round(0x95 + (0xff - 0x95) * t), 255];
}
export function renderIcon(size, ss = 4) {
  const buf = Buffer.alloc(size * size * 4);
  const n = ss * ss;
  for (let y = 0; y < size; y++) for (let x = 0; x < size; x++) {
    const acc = [0, 0, 0, 0];
    for (let sy = 0; sy < ss; sy++) for (let sx = 0; sx < ss; sx++) {
      const c = sample((x + (sx + 0.5) / ss) / size, (y + (sy + 0.5) / ss) / size);
      const a = c[3] / 255;
      acc[0] += c[0] * a; acc[1] += c[1] * a; acc[2] += c[2] * a; acc[3] += c[3];
    }
    const i = (y * size + x) * 4, aSum = acc[3] / 255;
    buf[i] = aSum ? Math.round(acc[0] / aSum) : 0;
    buf[i + 1] = aSum ? Math.round(acc[1] / aSum) : 0;
    buf[i + 2] = aSum ? Math.round(acc[2] / aSum) : 0;
    buf[i + 3] = Math.round(acc[3] / n);
  }
  return buf;
}
```

```js
function num(v) {
  const n = Number(v);
  return v !== '' && v != null && Number.isFinite(n) ? n : null;
}
function parseJson(text) {
  try { return JSON.parse(text); } catch { return null; }
}

export function parseWsFrame(text) {
  const j = parseJson(text);
  if (!j || !Array.isArray(j.data)) return [];
  const out = [];
  if (j.channel === 'token_stat') {
    for (const d of j.data) {
      const p = num(d && d.p);
      if (d && d.n === 'sol' && typeof d.a === 'string' && p > 0) {
        out.push({ type: 'price', mint: d.a, priceUsd: p, ts: num(d.t) ? d.t * 1000 : null });
      }
    }
  } else if (j.channel === 'major_coin_price') {
    for (const d of j.data) {
      const p = num(d && d.p);
      if (d && d.s === 'SOL' && p > 0) out.push({ type: 'sol', priceUsd: p, ts: num(d.ts) });
    }
  }
  return out;
}

export function parseInfoResponse(text) {
  const j = parseJson(text);
  if (!j || !Array.isArray(j.data)) return [];
  const out = [];
  for (const d of j.data) {
    if (!d || typeof d.address !== 'string' || typeof d.symbol !== 'string') continue;
    if (d.chain && d.chain !== 'sol') continue;
    const supply = num(d.total_supply);
    out.push({
      type: 'info', mint: d.address, symbol: d.symbol,
      name: typeof d.name === 'string' && d.name ? d.name : d.symbol,
      logo: typeof d.logo === 'string' && d.logo ? d.logo : null,
      decimals: num(d.decimals), totalSupply: supply > 0 ? supply : null,
    });
  }
  return out;
}

export function parseFeedMessage(msg) {
  if (!msg || typeof msg.data !== 'string') return [];
  if (msg.kind === 'ws') return parseWsFrame(msg.data);
  if (msg.kind === 'http') return parseInfoResponse(msg.data);
  return [];
}
```

```js
export const LIVE_TIMEOUT_MS = 3000;
export const STALE_MS = 5000;

export function initialPriceState(mint, now) {
  return { mint, status: 'waiting', priceUsd: null, source: null, lastLiveAt: 0, lastPriceAt: 0, since: now };
}

export function reducePrice(s, ev, now) {
  switch (ev.type) {
    case 'live':
      if (ev.mint !== s.mint) return s;
      return { ...s, status: 'live', priceUsd: ev.priceUsd, source: 'live', lastLiveAt: now, lastPriceAt: now };
    case 'backup':
      if (ev.mint !== s.mint || s.status === 'live') return s;
      return { ...s, status: 'backup', priceUsd: ev.priceUsd, source: 'backup', lastPriceAt: now };
    case 'backupFail':
      if (s.status === 'live' || s.status === 'waiting') return s;
      return s.priceUsd == null || now - s.lastPriceAt > STALE_MS ? { ...s, status: 'stale' } : s;
    case 'tick': {
      let n = s;
      if (n.status === 'live' && now - n.lastLiveAt > LIVE_TIMEOUT_MS) n = { ...n, status: 'backup' };
      if (n.status === 'waiting' && now - n.since > LIVE_TIMEOUT_MS) n = { ...n, status: 'backup' };
      if (n.status === 'backup' && n.priceUsd != null && now - n.lastPriceAt > STALE_MS) n = { ...n, status: 'stale' };
      return n;
    }
    default:
      return s;
  }
}

export const needsBackup = (s) => !!s && (s.status === 'backup' || s.status === 'stale');
export const canTrade = (s) => !!s && (s.status === 'live' || s.status === 'backup') && s.priceUsd > 0;
```

```js
export function quoteBuy({ balance, amountSol, priceSol, settings }) {
  const { platformFeePct, priorityFeeSol, slippagePct } = settings;
  if (!(amountSol > 0)) return { ok: false, error: 'Enter an amount above 0' };
  if (!(priceSol > 0)) return { ok: false, error: 'No price' };
  const cost = amountSol + priorityFeeSol;
  if (cost > balance + 1e-12) return { ok: false, error: 'Not enough SOL' };
  const platformFee = (amountSol * platformFeePct) / 100;
  const fillPriceSol = priceSol * (1 + slippagePct / 100);
  const tokens = (amountSol - platformFee) / fillPriceSol;
  return { ok: true, cost, tokens, platformFee, priorityFee: priorityFeeSol, fillPriceSol };
}

export function quoteSell({ tokensHeld, fraction, priceSol, settings }) {
  const { platformFeePct, priorityFeeSol, slippagePct } = settings;
  if (!(tokensHeld > 0)) return { ok: false, error: 'Nothing to sell' };
  if (!(fraction > 0 && fraction <= 1)) return { ok: false, error: 'Pick a % between 1 and 100' };
  if (!(priceSol > 0)) return { ok: false, error: 'No price' };
  const tokensOut = fraction === 1 ? tokensHeld : tokensHeld * fraction;
  const fillPriceSol = priceSol * (1 - slippagePct / 100);
  const gross = tokensOut * fillPriceSol;
  const platformFee = (gross * platformFeePct) / 100;
  const proceeds = Math.max(0, gross - platformFee - priorityFeeSol);
  return { ok: true, tokensOut, proceeds, platformFee, priorityFee: priorityFeeSol, fillPriceSol };
}

export function sellInitialsFraction({ invested, sold, tokensHeld, priceSol, settings }) {
  const target = invested - sold;
  if (!(target > 1e-9) || !(tokensHeld > 0) || !(priceSol > 0)) return null;
  const fillPriceSol = priceSol * (1 - settings.slippagePct / 100);
  const gross = (target + settings.priorityFeeSol) / (1 - settings.platformFeePct / 100);
  return Math.min(1, gross / (tokensHeld * fillPriceSol));
}
```

```js
export const DUST = 1e-9;

export function newPosition({ id, mint, coin, now }) {
  const c = coin || {};
  return {
    id, mint,
    symbol: c.symbol || '???',
    name: c.name || c.symbol || 'Unknown',
    logo: c.logo || null,
    totalSupply: c.totalSupply || null,
    openedAt: now, closedAt: null, status: 'open',
    invested: 0, sold: 0, tokens: 0, peakTokens: 0,
  };
}

export function applyBuy(pos, q) {
  const tokens = pos.tokens + q.tokens;
  return { ...pos, invested: pos.invested + q.cost, tokens, peakTokens: Math.max(pos.peakTokens, tokens) };
}

export function applySell(pos, q, now) {
  const left = Math.max(0, pos.tokens - q.tokensOut);
  const closed = left <= pos.peakTokens * DUST;
  return { ...pos, sold: pos.sold + q.proceeds, tokens: closed ? 0 : left, status: closed ? 'closed' : 'open', closedAt: closed ? now : null };
}

export function positionStats(pos, priceSol, settings) {
  const remaining = priceSol > 0 ? pos.tokens * priceSol * (1 - settings.slippagePct / 100) : 0;
  const pnlSol = pos.sold + remaining - pos.invested;
  const pnlPct = pos.invested > 0 ? (pnlSol / pos.invested) * 100 : 0;
  return { invested: pos.invested, sold: pos.sold, remaining, pnlSol, pnlPct };
}

export function openPositionFor(positions, mint) {
  return positions.find((p) => p.mint === mint && p.status === 'open') || null;
}

export function latestPositionFor(positions, mint) {
  return positions.filter((p) => p.mint === mint).sort((a, b) => b.openedAt - a.openedAt)[0] || null;
}
```

```js
export function makeLimit({ id, posId, mint, triggerMcUsd, fraction, currentMcUsd }) {
  if (!(triggerMcUsd > 0)) return { ok: false, error: 'Enter a market cap like 100k' };
  if (!(fraction > 0 && fraction <= 1)) return { ok: false, error: 'Pick a % between 1 and 100' };
  if (!(currentMcUsd > 0)) return { ok: false, error: 'No price' };
  if (triggerMcUsd === currentMcUsd) return { ok: false, error: 'Target equals current market cap' };
  return { ok: true, limit: { id, posId, mint, triggerMcUsd, fraction, direction: triggerMcUsd > currentMcUsd ? 'above' : 'below' } };
}

export function isTriggered(limit, mcUsd) {
  return limit.direction === 'above' ? mcUsd >= limit.triggerMcUsd : mcUsd <= limit.triggerMcUsd;
}

export function triggeredLimits(limits, mint, mcUsd) {
  if (!(mcUsd > 0)) return [];
  return limits.filter((l) => l.mint === mint && isTriggered(l, mcUsd));
```

```js
export const MAX_TICKS = 5000;

export function addTick(ticks, tick, max = MAX_TICKS) {
  const last = ticks[ticks.length - 1];
  if (last && tick[0] <= last[0]) return ticks;
  const next = [...ticks, tick];
  if (next.length <= max) return next;
  return next.filter((_, i) => i % 2 === 0 || i === next.length - 1);
}

export function findGaps(ticks, from, to, gapMs = 30000) {
  const pts = [from, ...ticks.map((t) => t[0]).filter((t) => t > from && t < to), to];
  const gaps = [];
  for (let i = 1; i < pts.length; i++) if (pts[i] - pts[i - 1] > gapMs) gaps.push([pts[i - 1], pts[i]]);
  return gaps;
}

export function mergeSeries(a, b) {
  const seen = new Set();
  return [...a, ...b]
    .filter(([t]) => (seen.has(t) ? false : seen.add(t)))
    .sort((x, y) => x[0] - y[0]);
}

export function lastPrice(ticks) {
  return ticks.length ? ticks[ticks.length - 1][1] : null;
}
```

```js
function upsert(list, pos) {
  const i = list.findIndex((p) => p.id === pos.id);
  if (i < 0) return [...list, pos];
  const copy = list.slice();
  copy[i] = pos;

function refreshCoin(pos, coin) {
  return {
    ...pos,
    symbol: pos.symbol === '???' && coin.symbol ? coin.symbol : pos.symbol,
    name: pos.name === 'Unknown' && (coin.name || coin.symbol) ? coin.name || coin.symbol : pos.name,
    logo: pos.logo || coin.logo || null,
    totalSupply: pos.totalSupply || coin.totalSupply || null,
  };

export function executeTrade(ledger, cmd, ctx) {
  const { priceUsd, solUsd, settings, now } = ctx;
  if (!(priceUsd > 0) || !(solUsd > 0)) return { ok: false, error: 'No price' };
  const priceSol = priceUsd / solUsd;
  let pos = openPositionFor(ledger.positions, cmd.mint);
  let quote;
  let side;
  if (cmd.side === 'buy') {
    quote = quoteBuy({ balance: ledger.balance, amountSol: cmd.amountSol, priceSol, settings });
    if (!quote.ok) return quote;
    const coin = cmd.coin || {};
    pos = pos ? refreshCoin(pos, coin) : newPosition({ id: ctx.newId(), mint: cmd.mint, coin, now });
    pos = applyBuy(pos, quote);
    side = 'buy';
  } else {
    if (!pos) return { ok: false, error: 'No open position' };
    let fraction = cmd.fraction;
    if (cmd.side === 'sellInit') {
      fraction = sellInitialsFraction({ invested: pos.invested, sold: pos.sold, tokensHeld: pos.tokens, priceSol, settings });
      if (fraction == null) return { ok: false, error: 'Initials already taken out' };
    }
    quote = quoteSell({ tokensHeld: pos.tokens, fraction, priceSol, settings });
    if (!quote.ok) return quote;
    pos = applySell(pos, quote, now);
    side = 'sell';
  }
  const trade = {
    id: ctx.newId(), posId: pos.id, mint: cmd.mint, symbol: pos.symbol, side, ts: now,
    amountSol: side === 'buy' ? quote.cost : quote.proceeds,
    tokens: side === 'buy' ? quote.tokens : quote.tokensOut,
    priceUsd, solUsd, mcUsd: priceUsd * (pos.totalSupply || DEFAULT_SUPPLY),
    fillPriceSol: quote.fillPriceSol, platformFee: quote.platformFee, priorityFee: quote.priorityFee,
    source: ctx.source || 'live', reason: cmd.reason || null,
  };
  const balance = side === 'buy' ? ledger.balance - quote.cost : ledger.balance + quote.proceeds;
  const limits = pos.status === 'closed' ? ledger.limits.filter((l) => l.posId !== pos.id) : ledger.limits;
  return { ok: true, trade, ledger: { ...ledger, balance, positions: upsert(ledger.positions, pos), trades: [...ledger.trades, trade], limits } };
}

export function addLimit(ledger, args) {
  const pos = ledger.positions.find((p) => p.id === args.posId && p.status === 'open');
  if (!pos) return { ok: false, error: 'No open position' };
  const r = makeLimit(args);
  if (!r.ok) return r;
  return { ok: true, limit: r.limit, ledger: { ...ledger, limits: [...ledger.limits, r.limit] } };
}

export function cancelLimit(ledger, id) {
  return { ok: true, ledger: { ...ledger, limits: ledger.limits.filter((l) => l.id !== id) } };
}

export function fireLimit(ledger, limitId, ctx) {
  const limit = ledger.limits.find((l) => l.id === limitId);
  if (!limit) return { ok: false, error: 'Limit already gone' };
  const r = executeTrade(ledger, { side: 'sell', mint: limit.mint, fraction: limit.fraction, reason: 'limit' }, ctx);
  if (!r.ok) return r;
  return { ...r, ledger: { ...r.ledger, limits: r.ledger.limits.filter((l) => l.id !== limitId) } };
}

export function resetRun(ledger, settings, now) {
  const run = { startedAt: ledger.runStartedAt, endedAt: now, endBalance: ledger.balance, trades: ledger.trades, positions: ledger.positions };
  return { ...initialLedger(settings, now), pastRuns: [...ledger.pastRuns, run] };
}
```

```js
export const SOL_MINT = 'So11111111111111111111111111111111111111112';

export const jupiterUrl = (mints) => `https://lite-api.jup.ag/price/v3?ids=${mints.join(',')}`;
export const dexScreenerUrl = (mint) => `https://api.dexscreener.com/tokens/v1/solana/${mint}`;
export const geckoPoolsUrl = (mint) => `https://api.geckoterminal.com/api/v2/networks/solana/tokens/${mint}/pools?page=1`;
export const geckoOhlcvUrl = (pool, mint, beforeSec) =>
  `https://api.geckoterminal.com/api/v2/networks/solana/pools/${pool}/ohlcv/minute?aggregate=1&limit=1000&currency=usd&token=${mint}&before_timestamp=${beforeSec}`;

export function parseJupiter(json, mint) {
  const p = Number(json && json[mint] && json[mint].usdPrice);
  return p > 0 ? p : null;
}

export function parseDexScreener(json, mint) {
  if (!Array.isArray(json)) return null;
  const pairs = json.filter((p) => p && p.baseToken && p.baseToken.address === mint && Number(p.priceUsd) > 0);
  pairs.sort((a, b) => ((b.liquidity && b.liquidity.usd) || 0) - ((a.liquidity && a.liquidity.usd) || 0));
  return pairs.length ? Number(pairs[0].priceUsd) : null;
}

export function parseGeckoPools(json) {
  const d = json && json.data;
  if (!Array.isArray(d) || !d.length) return null;
  return (d[0].attributes && d[0].attributes.address) || null;
}

export function parseGeckoOhlcv(json) {
  const list = json && json.data && json.data.attributes && json.data.attributes.ohlcv_list;
  if (!Array.isArray(list)) return [];
  return list.map((r) => [r[0] * 1000, Number(r[4])]).filter(([t, p]) => t > 0 && p > 0).sort((a, b) => a[0] - b[0]);
}

export async function fetchJson(url, timeoutMs = 4000) {
  const ac = new AbortController();
  const timer = setTimeout(() => ac.abort(), timeoutMs);
  try {
    const r = await fetch(url, { signal: ac.signal });
    if (!r.ok) throw new Error(`HTTP ${r.status}`);
    return await r.json();
  } finally {
    clearTimeout(timer);
  }
}

export async function backupPrice(mint, getJson = fetchJson) {
  try {
    const j = await getJson(jupiterUrl([mint, SOL_MINT]));
    const p = parseJupiter(j, mint);
    if (p) return { priceUsd: p, solUsd: parseJupiter(j, SOL_MINT), source: 'jupiter' };
  } catch (e) {
    console.warn('[SolSim] jupiter', e.message);
  }
  try {
    const p = parseDexScreener(await getJson(dexScreenerUrl(mint)), mint);
    if (p) return { priceUsd: p, solUsd: null, source: 'dexscreener' };
  } catch (e) {
    console.warn('[SolSim] dexscreener', e.message);
  }
  return null;
}

export async function priceHistory(mint, fromMs, toMs, getJson = fetchJson) {
  try {
    const pool = parseGeckoPools(await getJson(geckoPoolsUrl(mint)));
    if (!pool) return [];
    const pts = parseGeckoOhlcv(await getJson(geckoOhlcvUrl(pool, mint, Math.ceil(toMs / 1000) + 60)));
    return pts.filter(([t]) => t >= fromMs - 60000 && t <= toMs + 60000);
  } catch (e) {
    console.warn('[SolSim] geckoterminal', e.message);
    return [];
  }
}

export function bytesToDataUrl(bytes, mime) {
  let bin = '';
  for (let i = 0; i < bytes.length; i += 0x8000) bin += String.fromCharCode.apply(null, bytes.subarray(i, i + 0x8000));
  return `data:${mime};base64,${btoa(bin)}`;
}
```

```js
export function createView(h) {
  const host = document.createElement('div');
  host.style.cssText = 'position:fixed;z-index:2147483000;left:16px;top:140px;display:none;';
  const root = host.attachShadow({ mode: 'closed' });
  const link = document.createElement('link');
  link.rel = 'stylesheet';
  link.href = chrome.runtime.getURL('src/panel/panel.css');
  const tpl = document.createElement('template');
  tpl.innerHTML = HTML.trim();
  root.append(link, tpl.content);
  document.documentElement.append(host);
  const $ = (s) => root.querySelector(s);

  let collapsed = false;
  let lastVm = null;
  let presetKey = '';
  let toastTimer = 0;

  function place(x, y) {
    const w = host.offsetWidth || 290;
    host.style.left = Math.min(Math.max(0, x), window.innerWidth - w) + 'px';
    host.style.top = Math.min(Math.max(0, y), window.innerHeight - 40) + 'px';
  }
  function setCollapsed(v, save) {
    collapsed = v;
    $('.body').hidden = v;
    $('[data-act=collapse]').textContent = v ? '▸' : '▾';
    if (save) chrome.storage.local.set({ panelCollapsed: v });
  }
  chrome.storage.local.get(['panelPos', 'panelCollapsed']).then((o) => {
    if (o.panelPos) place(o.panelPos.x, o.panelPos.y);
    setCollapsed(!!o.panelCollapsed, false);
  });

  const handle = $('[data-drag]');
  let drag = null;
  handle.addEventListener('pointerdown', (e) => {
    if (e.target.closest('button')) return;
    const r = host.getBoundingClientRect();
    drag = { dx: e.clientX - r.left, dy: e.clientY - r.top };
    handle.setPointerCapture(e.pointerId);
  });
  handle.addEventListener('pointermove', (e) => { if (drag) place(e.clientX - drag.dx, e.clientY - drag.dy); });
  handle.addEventListener('pointerup', () => {
    if (!drag) return;
    drag = null;
    chrome.storage.local.set({ panelPos: { x: parseFloat(host.style.left), y: parseFloat(host.style.top) } });
  });

  // Keep GMGN hotkeys from firing while typing in the panel.
  for (const t of ['keydown', 'keyup', 'keypress']) root.addEventListener(t, (e) => e.stopPropagation());

  root.addEventListener('click', (e) => {
    const b = e.target.closest('button');
    if (!b || b.disabled) return;
    if (b.dataset.buy) return h.onBuy(Number(b.dataset.buy));
    if (b.dataset.sell) return h.onSell(Number(b.dataset.sell));
    if (b.dataset.cancel) return h.onLimitCancel(b.dataset.cancel);
    switch (b.dataset.act) {
      case 'settings': return h.onSettings();
      case 'collapse': return setCollapsed(!collapsed, true);
      case 'buycustom': return h.onBuy(Number($('[data-buyamt]').value));
      case 'sellcustom': return h.onSell(Number($('[data-sellpct]').value));
      case 'sellinit': return h.onSellInit();

      case 'limitsave':
        return h.onLimitAdd({ mcUsd: parseCompact($('[data-limmc]').value), pct: Number($('[data-limpct]').value) });
      case 'card': return lastVm && lastVm.cardPosId && h.onCard(lastVm.cardPosId);
    }
  });

      b.className = kind;
      b.dataset[kind] = String(v);
      b.textContent = label(v);
      return b;
    }));

      const row = document.createElement('div');

      const t = document.createElement('span');
      t.textContent = l.text;
      const x = document.createElement('button');

      x.dataset.cancel = l.id;
      row.append(t, x);
```

```js
const send = (msg) =>
  chrome.runtime.sendMessage(msg).catch((e) => ({ ok: false, error: String((e && e.message) || e) }));

export function start() {
  const st = {
    mint: null, price: null, solUsd: null, infos: new Map(),
    ledger: null, settings: DEFAULT_SETTINGS,
    firing: new Map(), lastSlow: 0, solSavedAt: 0,
  };

  const coin = () => {
    const i = st.infos.get(st.mint) || {};
    return { symbol: i.symbol || symbolFromTitle(document.title) || '???', name: i.name || null, logo: i.logo || null, totalSupply: i.totalSupply || null };
  };

  const supply = () => coin().totalSupply || (openPos() && openPos().totalSupply) || DEFAULT_SUPPLY;
  const market = () => ({ priceUsd: st.price.priceUsd, solUsd: st.solUsd, source: st.price.source });
  const tradable = () => !!st.price && canTrade(st.price) && st.solUsd > 0;

  async function trade(cmd) {
    if (!tradable()) return view.toast('No price yet', 'err');
    const r = await send({ cmd: 'trade', trade: { ...cmd, mint: st.mint, coin: coin() }, market: market() });
    if (!r || !r.ok) return view.toast((r && r.error) || 'Trade failed', 'err');
    const t = r.trade;
    view.toast(t.side === 'buy' ? `Bought ${fmtSol(t.amountSol)} SOL of ${t.symbol}` : `Sold for ${fmtSol(t.amountSol)} SOL`, 'ok');

    onBuy: (amountSol) => trade({ side: 'buy', amountSol }),
    onSell: (pct) => trade({ side: 'sell', fraction: pct / 100 }),
    onSellInit: () => trade({ side: 'sellInit' }),
    onLimitAdd: async ({ mcUsd, pct }) => {
      const pos = openPos();
      if (!pos || !tradable()) return view.toast('No open position', 'err');
      const r = await send({ cmd: 'limitAdd', limit: { posId: pos.id, mint: st.mint, triggerMcUsd: mcUsd, fraction: pct / 100, currentMcUsd: st.price.priceUsd * supply() } });
      if (!r.ok) return view.toast(r.error, 'err');
      view.clearLimitForm();
      view.toast(`Limit set at ${fmtCompactUsd(mcUsd)}`, 'ok');
    },
    onLimitCancel: (limitId) => send({ cmd: 'limitCancel', limitId }),
    onSettings: () => send({ cmd: 'openSettings' }),
    onCard: (posId) => send({ cmd: 'openCard', posId }),

    if (!p) return { status: 'waiting', text: '' };
    if (p.status === 'live') return st.solUsd ? { status: 'live', text: 'Live' } : { status: 'waiting', text: 'Waiting for SOL price…' };
    if (p.status === 'backup') return p.priceUsd ? { status: 'backup', text: 'Backup price' } : { status: 'waiting', text: 'Finding price…' };
    if (p.status === 'stale') return { status: 'stale', text: 'No price' };
    return { status: 'waiting', text: 'Waiting for price…' };
  }

  function render() {
    if (raf) return;
    raf = requestAnimationFrame(() => {
      raf = 0;
      const pos = openPos();
      const priceSol = st.price && st.price.priceUsd && st.solUsd ? st.price.priceUsd / st.solUsd : 0;
      const stats = pos ? positionStats(pos, priceSol, st.settings) : null;
      const cardPos = pos || (st.ledger && st.mint ? latestPositionFor(st.ledger.positions, st.mint) : null);
      view.render({
        visible: !!st.mint,
        balance: st.ledger ? st.ledger.balance : 0,
        symbol: coin().symbol,
        badge: badgeFor(st.price),
        mcUsd: st.price && st.price.priceUsd ? st.price.priceUsd * supply() : null,
        mcEstimated: !coin().totalSupply && !(pos && pos.totalSupply),
        canTrade: tradable(),
        buyPresets: st.settings.buyPresets,
        sellPresets: st.settings.sellPresets,
        pos: stats,
        canSellInit: !!pos && pos.invested - pos.sold > 1e-9 && pos.tokens > 0,
        limits: ((st.ledger && st.ledger.limits) || []).filter((l) => l.mint === st.mint).map((l) => ({
          id: l.id,
          text: `Sell ${Math.round(l.fraction * 100)}% ${l.direction === 'above' ? '≥' : '≤'} ${fmtCompactUsd(l.triggerMcUsd)}`,

        cardPosId: cardPos ? cardPos.id : null,
      });
    });

    if (!st.ledger || !tradable()) return;
    const mc = st.price.priceUsd * supply();
    const now = Date.now();
    for (const l of triggeredLimits(st.ledger.limits, st.mint, mc)) {
      if (now - (st.firing.get(l.id) || 0) < 5000) continue;
      st.firing.set(l.id, now);

        if (r && r.ok) view.toast(`Limit hit: sold ${Math.round(l.fraction * 100)}%`, 'ok');
      });
    }
  }

  window.addEventListener('message', (ev) => {
    if (ev.source !== window || !ev.data || ev.data.source !== 'solsim-feed') return;
    let priced = false;
    for (const e of parseFeedMessage(ev.data)) {
      if (e.type === 'info') st.infos.set(e.mint, e);
      else if (e.type === 'sol') {
        st.solUsd = e.priceUsd;
        const now = Date.now();
        if (now - st.solSavedAt > 30000) { st.solSavedAt = now; chrome.storage.local.set({ solUsd: e.priceUsd }); }
      } else if (e.type === 'price' && st.price && e.mint === st.mint) {
        st.price = reducePrice(st.price, { type: 'live', mint: e.mint, priceUsd: e.priceUsd }, Date.now());
        priced = true;
      }
    }
    if (priced) checkLimits();
    render();
  });
  window.postMessage({ source: 'solsim-panel', kind: 'hello' }, location.origin);

  chrome.storage.onChanged.addListener((ch, area) => {
    if (area !== 'local') return;
    if (ch.ledger) st.ledger = ch.ledger.newValue;
    if (ch.settings) st.settings = { ...DEFAULT_SETTINGS, ...ch.settings.newValue };
    if (ch.ledger || ch.settings) render();
  });
  send({ cmd: 'getState' }).then((r) => {
    if (r && r.ledger) { st.ledger = r.ledger; st.settings = r.settings; }
    render();
  });
  chrome.storage.local.get('solUsd').then((o) => { if (o.solUsd && !st.solUsd) st.solUsd = o.solUsd; });

  async function slow(now) {
    const mint = st.mint;
    if (needsBackup(st.price) || !st.solUsd) {
      const r = await send({ cmd: 'backupPrice', mint });
      if (mint !== st.mint) return;
      if (r && r.ok) {
        if (r.solUsd && !st.solUsd) st.solUsd = r.solUsd;
        st.price = reducePrice(st.price, { type: 'backup', mint, priceUsd: r.priceUsd }, Date.now());
        checkLimits();
      } else {
        st.price = reducePrice(st.price, { type: 'backupFail' }, Date.now());
      }
    }
    const pos = openPos();
    if (pos && canTrade(st.price)) send({ cmd: 'tick', posId: pos.id, ts: now, priceUsd: st.price.priceUsd });
    render();
  }

  const timer = setInterval(() => {
    if (!chrome.runtime || !chrome.runtime.id) {
      clearInterval(timer);
      view.destroy();
      return;
    }
    const now = Date.now();
    const m = mintFromPath(location.pathname);
    if (m !== st.mint) {
      st.mint = m;
      st.price = m ? initialPriceState(m, now) : null;
      view.clearLimitForm();
    }
    if (st.price) st.price = reducePrice(st.price, { type: 'tick' }, now);
    if (st.mint && now - st.lastSlow >= 2000) {
      st.lastSlow = now;
      slow(now);
    }
    render();
  }, 500);
}
```

```js
export function loadImage(src) {
  return new Promise((resolve, reject) => {
    const img = new Image();
    img.onload = () => resolve(img);
    img.onerror = () => reject(new Error('image failed to load'));
    img.src = src;
  });
}

export async function downscaleToDataUrl(file, maxW, maxH, type = 'image/jpeg', quality = 0.9) {
  const url = URL.createObjectURL(file);
  try {
    const img = await loadImage(url);
    const k = Math.min(1, maxW / img.width, maxH / img.height);
    const c = document.createElement('canvas');
    c.width = Math.max(1, Math.round(img.width * k));
    c.height = Math.max(1, Math.round(img.height * k));
    c.getContext('2d').drawImage(img, 0, 0, c.width, c.height);
    return c.toDataURL(type, quality);
  } finally {
    URL.revokeObjectURL(url);
  }
}
```

```js
import { fmtUsd, fmtSol, fmtPct } from '../shared/format.js';
import { mergeSeries } from '../engine/ticks.js';

export const SIZES = { landscape: { w: 1600, h: 900 }, portrait: { w: 1080, h: 1350 } };

export function cardModel({ position, trades, series = [], lastPriceUsd, solUsdNow, currency = 'USD', displayName = '', settings }) {
  const own = trades.filter((t) => t.posId === position.id).sort((a, b) => a.ts - b.ts);
  const sum = (side, f) => own.filter((t) => t.side === side).reduce((acc, t) => acc + f(t), 0);
  const boughtSol = sum('buy', (t) => t.amountSol);
  const soldSol = sum('sell', (t) => t.amountSol);
  const boughtUsd = sum('buy', (t) => t.amountSol * t.solUsd);
  const soldUsd = sum('sell', (t) => t.amountSol * t.solUsd);
  const solNow = solUsdNow > 0 ? solUsdNow : (own.length ? own[own.length - 1].solUsd : 0);
  const holdSol = position.tokens > 0 && lastPriceUsd > 0 && solNow > 0
    ? position.tokens * (lastPriceUsd / solNow) * (1 - settings.slippagePct / 100) : 0;
  const usd = currency === 'USD';
  const bought = usd ? boughtUsd : boughtSol;
  const sold = usd ? soldUsd : soldSol;
  const hold = usd ? holdSol * solNow : holdSol;
  const pnl = sold + hold - bought;
  const pct = bought > 0 ? (pnl / bought) * 100 : 0;
  const fmt = (n) => (usd ? fmtUsd(n) : `${fmtSol(n)} SOL`);
  const positive = pnl >= 0;
  const markers = own.map((t) => ({ t: t.ts, p: t.priceUsd, side: t.side === 'buy' ? 'B' : 'S' }));
  return {
    symbol: position.symbol, logo: position.logo || null, currency, positive,
    pnlText: (pnl > 0 ? '+' : '') + fmt(pnl),
    pctText: fmtPct(pct),
    stats: [
      { label: 'PNL', value: `${fmt(pnl)} (${fmtPct(pct)})`, tone: positive ? 'up' : 'down' },
      { label: 'Sold', value: fmt(sold) },
      { label: 'Hold', value: fmt(hold) },
      { label: 'Bought', value: fmt(bought) },
    ],
    series: series.length >= 2 ? mergeSeries(series, markers.map((m) => [m.t, m.p])) : [],
    markers, displayName, isOpen: position.status === 'open',
  };
}

export function chartGeometry(series, markers, box, pad = 0.08) {
  if (!series || series.length < 2) return null;
  const t0 = series[0][0];
  const t1 = series[series.length - 1][0];
  let lo = Infinity;
  let hi = -Infinity;
  for (const [, p] of series) { lo = Math.min(lo, p); hi = Math.max(hi, p); }
  for (const m of markers) { lo = Math.min(lo, m.p); hi = Math.max(hi, m.p); }
  if (hi === lo) { hi = lo * 1.01 + 1e-12; lo *= 0.99; }
  const span = hi - lo;
  lo -= span * pad;
  hi += span * pad;
  const tx = (t) => box.x + (t1 === t0 ? 0 : (t - t0) / (t1 - t0)) * box.w;
  const py = (p) => box.y + (1 - (p - lo) / (hi - lo)) * box.h;
  return {
    path: series.map(([t, p]) => [tx(t), py(p)]),
    markers: markers.filter((m) => m.t >= t0 && m.t <= t1).map((m) => ({ x: tx(m.t), y: py(m.p), side: m.side })),
  };
}

export function shareText(model) {
  return `${model.pctText} on $${model.symbol} ${model.positive ? '📈' : '📉'} (paper trade via SolSim)`;
}
```

```js
  return function () {
    seed |= 0;
    seed = (seed + 0x6d2b79f5) | 0;
```

```js
  const setFont = () => (ctx.font = `${o.weight || 400} ${size}px ${FONT}`);
  setFont();
  if (o.maxWidth) while (size > 12 && ctx.measureText(str).width > o.maxWidth) { size -= 2; setFont(); }
  ctx.textAlign = o.align || 'left';
  ctx.textBaseline = 'alphabetic';
  ctx.fillStyle = o.color || '#fff';
  if (o.glow) {
    ctx.save();
    ctx.shadowColor = o.glow;
    ctx.shadowBlur = 40; ctx.fillText(str, x, y);
    ctx.shadowBlur = 16; ctx.fillText(str, x, y);
    ctx.restore();
  }
  ctx.fillText(str, x, y);
}

function circleImage(ctx, img, x, y, s) {
  ctx.save();
  ctx.beginPath(); ctx.arc(x + s / 2, y + s / 2, s / 2, 0, Math.PI * 2); ctx.clip();
  ctx.drawImage(img, x, y, s, s);
  ctx.restore();
}

function drawCoin(ctx, img, symbol, x, y, s) {
  if (img) return circleImage(ctx, img, x, y, s);
  ctx.fillStyle = '#2a2f3a';
  ctx.beginPath(); ctx.arc(x + s / 2, y + s / 2, s / 2, 0, Math.PI * 2); ctx.fill();
  text(ctx, (symbol || '?').slice(0, 1).toUpperCase(), x + s / 2, y + s * 0.68, { size: s * 0.5, weight: 800, align: 'center' });
}

function drawBackground(ctx, bg, w, h) {
  if (bg && bg.kind === 'upload' && bg.img) {
    const k = Math.max(w / bg.img.width, h / bg.img.height);
    const dw = bg.img.width * k, dh = bg.img.height * k;
    ctx.drawImage(bg.img, (w - dw) / 2, (h - dh) / 2, dw, dh);
    return;
  }
  const b = BACKGROUNDS.find((x) => bg && x.id === bg.id) || BACKGROUNDS[0];
  b.draw(ctx, w, h);
}

function drawChart(ctx, model, box, neon) {
  const g = chartGeometry(model.series, model.markers, box);
  if (!g) return;
  ctx.save();
  ctx.lineJoin = 'round'; ctx.lineCap = 'round'; ctx.strokeStyle = '#fff'; ctx.lineWidth = 4;
  if (neon) { ctx.shadowColor = 'rgba(255,255,255,0.9)'; ctx.shadowBlur = 18; }
  ctx.beginPath();
  g.path.forEach(([x, y], i) => (i ? ctx.lineTo(x, y) : ctx.moveTo(x, y)));
  ctx.stroke();
  ctx.restore();
  for (const m of g.markers) {
    const c = m.side === 'B' ? GREEN : RED;
    ctx.save();
    ctx.beginPath(); ctx.arc(m.x, m.y, 16, 0, Math.PI * 2); ctx.fillStyle = c;
    if (neon) { ctx.shadowColor = c; ctx.shadowBlur = 20; }
    ctx.fill();
    ctx.restore();
    ctx.font = `800 18px ${FONT}`; ctx.fillStyle = '#0b0f14'; ctx.textAlign = 'center'; ctx.textBaseline = 'middle';
    ctx.fillText(m.side, m.x, m.y + 1);
  }
}

export function drawCard(ctx, model, opts, assets = {}) {
  const { w, h } = SIZES[opts.layout];
  const L = LAYOUTS[opts.layout];
  ctx.save();
  ctx.clearRect(0, 0, w, h);
  drawBackground(ctx, opts.background, w, h);
  if (opts.overlay) { ctx.fillStyle = 'rgba(0,0,0,0.45)'; ctx.fillRect(0, 0, w, h); }
  const [bx, by, bw, bh] = L.band;
  const band = ctx.createLinearGradient(0, by, 0, by + bh);
  band.addColorStop(0, 'rgba(0,0,0,0.25)'); band.addColorStop(1, 'rgba(0,0,0,0.6)');
  ctx.fillStyle = band; ctx.fillRect(bx, by, bw, bh);

  drawLogoMark(ctx, L.logo[0], L.logo[1], L.logo[2]);
  text(ctx, 'SolSim', L.brand[0], L.brand[1], { size: 44, weight: 800 });
  text(ctx, 'PAPER TRADE', L.tag[0], L.tag[1], { size: 26, weight: 700, color: 'rgba(255,255,255,0.7)', align: 'right' });

  drawCoin(ctx, assets.coinImg, model.symbol, L.coin[0], L.coin[1], L.coin[2]);
  text(ctx, model.symbol, L.sym[0], L.sym[1], { size: 52, weight: 800, maxWidth: 600 });

  const col = model.positive ? GREEN : RED;
  text(ctx, model.pnlText, L.pnl[0], L.pnl[1], { size: L.pnl[2], weight: 900, color: col, glow: opts.neon ? col : null, maxWidth: L.pnl[3] });
  text(ctx, model.pctText, L.pct[0], L.pct[1], { size: 44, weight: 700, color: col });

  drawChart(ctx, model, L.chart, opts.neon);

  const s = L.stats;
  const cells = [[s.x1, s.v1, s.y1], [s.x2, s.v2, s.y1], [s.x1, s.v1, s.y2], [s.x2, s.v2, s.y2]];
  model.stats.forEach((st, i) => {
    const [x, vx, y] = cells[i];
    text(ctx, st.label, x, y, { size: 34, weight: 600, color: 'rgba(255,255,255,0.6)' });
    text(ctx, st.value, vx, y, {
      size: 38, weight: 800,
      color: st.tone === 'up' ? GREEN : st.tone === 'down' ? RED : '#fff',
      maxWidth: i % 2 === 0 ? s.x2 - s.v1 - 24 : w - s.v2 - 48,
    });
  });

  const [nx, ny, ns] = L.name;
  let tx = nx;
  if (assets.avatarImg) { circleImage(ctx, assets.avatarImg, nx, ny - ns * 0.75, ns); tx += ns + 16; }
  if (model.displayName) text(ctx, model.displayName, tx, ny, { size: 34, weight: 700 });
  text(ctx, model.isOpen ? 'Position open · SolSim' : 'SolSim', L.foot[0], L.foot[1], { size: 26, weight: 600, color: 'rgba(255,255,255,0.55)', align: 'right' });
  ctx.restore();
}
```

```js
import { cardModel, SIZES, shareText } from './model.js';
import { drawCard } from './render.js';
import { BACKGROUNDS } from './backgrounds.js';
import { findGaps, mergeSeries, lastPrice } from '../engine/ticks.js';
import { downscaleToDataUrl, loadImage } from '../shared/image.js';

const $ = (s) => document.querySelector(s);
const send = (m) => chrome.runtime.sendMessage(m);
const DEFAULT_PREFS = { layout: 'landscape', neon: true, overlay: true, currency: null, bg: { kind: 'builtin', id: 'starfield' } };
const MAX_UPLOADS = 5;
const MAX_UPLOAD_CHARS = Math.round(2 * 1024 * 1024 * 1.37);
const S = { assets: { uploads: [] } };

const status = (t) => { $('#status').textContent = t; };
const savePrefs = () => chrome.storage.local.set({ cardPrefs: S.prefs });

async function init() {
  const posId = new URLSearchParams(location.search).get('pos');
  const { ledger, settings } = await send({ cmd: 'getState' });
  const position = ledger.positions.find((p) => p.id === posId);
  if (!position) { $('#msg').textContent = 'Position not found (it may have been reset).'; return; }
  const tickKey = 'ticks:' + posId;
  const o = await chrome.storage.local.get([tickKey, 'solUsd', 'cardPrefs', 'cardUploads']);
  const ticks = o[tickKey] || [];
  const trades = ledger.trades.filter((t) => t.posId === posId);
  const last = trades[trades.length - 1];
  Object.assign(S, {
    position, settings, trades, ticks, series: ticks,
    solUsd: o.solUsd || (last && last.solUsd) || null,
    lastPriceUsd: lastPrice(ticks) ?? (last ? last.priceUsd : null),
    prefs: { ...DEFAULT_PREFS, ...(o.cardPrefs || {}) },
    uploads: o.cardUploads || [],
  });
  if (!S.prefs.currency) S.prefs.currency = settings.pnlCurrency;
  document.title = `SolSim · ${position.symbol}`;
  await loadAssets();
  bindControls();
  draw();

  const end = position.closedAt || Date.now();
  const gaps = findGaps(ticks, position.openedAt, end);
  if (gaps.length) {
    status('Loading price history…');
    const r = await send({ cmd: 'history', mint: position.mint, from: position.openedAt, to: end });
    const extra = r && r.ok ? r.points.filter(([t]) => gaps.some(([a, b]) => t > a && t < b)) : [];
    if (extra.length) { S.series = mergeSeries(ticks, extra); draw(); }
    status('');
  }
}

async function safeImage(src) {
  try { return src ? await loadImage(src) : null; } catch { return null; }
}

async function loadAssets() {
  if (S.position.logo) {
    const r = await send({ cmd: 'logoData', url: S.position.logo });
    S.assets.coin = r && r.ok ? await safeImage(r.dataUrl) : null;
  }
  S.assets.avatar = await safeImage(S.settings.avatar);
  S.assets.uploads = await Promise.all(S.uploads.map(safeImage));
}

function background() {
  const bg = S.prefs.bg;
  if (bg.kind === 'upload' && S.assets.uploads[bg.index]) return { kind: 'upload', img: S.assets.uploads[bg.index] };
  return { kind: 'builtin', id: bg.kind === 'builtin' ? bg.id : 'starfield' };
}

function draw() {
  const { w, h } = SIZES[S.prefs.layout];
  const cv = $('#cv');
  cv.width = w; cv.height = h;
  cv.className = S.prefs.layout;
  S.model = cardModel({
    position: S.position, trades: S.trades, series: S.series, lastPriceUsd: S.lastPriceUsd,
    solUsdNow: S.solUsd, currency: S.prefs.currency, displayName: S.settings.displayName, settings: S.settings,
  });
  drawCard(cv.getContext('2d'), S.model, { layout: S.prefs.layout, neon: S.prefs.neon, overlay: S.prefs.overlay, background: background() },
    { coinImg: S.assets.coin, avatarImg: S.assets.avatar });
  syncControls();
}

function syncControls() {
  $('#neon').checked = S.prefs.neon;
  $('#overlay').checked = S.prefs.overlay;
  for (const b of document.querySelectorAll('#layout button')) b.classList.toggle('on', b.dataset.v === S.prefs.layout);
  for (const b of document.querySelectorAll('#currency button')) b.classList.toggle('on', b.dataset.v === S.prefs.currency);
  for (const t of document.querySelectorAll('#bgs .thumb')) {
    const bg = S.prefs.bg;
    const on = t.dataset.kind === bg.kind && (bg.kind === 'builtin' ? t.dataset.id === bg.id : Number(t.dataset.index) === bg.index);
    t.classList.toggle('on', on);
  }
}

function renderThumbs() {
  const thumbs = BACKGROUNDS.map((b) => {
    const c = document.createElement('canvas');
    c.width = 160; c.height = 90; c.className = 'thumb'; c.title = b.name;
    c.dataset.kind = 'builtin'; c.dataset.id = b.id;
    b.draw(c.getContext('2d'), 160, 90);
    return c;
  });
  S.assets.uploads.forEach((img, i) => {
    if (!img) return;
    const wrap = document.createElement('div');
    wrap.className = 'thumb upl'; wrap.dataset.kind = 'upload'; wrap.dataset.index = String(i);
    const im = document.createElement('img'); im.src = img.src;
    const x = document.createElement('button'); x.textContent = '✕'; x.className = 'del'; x.dataset.del = String(i);
    wrap.append(im, x);
    thumbs.push(wrap);
  });
  $('#bgs').replaceChildren(...thumbs);
}

function bindControls() {
  renderThumbs();
  $('#neon').onchange = (e) => { S.prefs.neon = e.target.checked; savePrefs(); draw(); };
  $('#overlay').onchange = (e) => { S.prefs.overlay = e.target.checked; savePrefs(); draw(); };
  $('#layout').onclick = (e) => { const v = e.target.dataset.v; if (v) { S.prefs.layout = v; savePrefs(); draw(); } };
  $('#currency').onclick = (e) => { const v = e.target.dataset.v; if (v) { S.prefs.currency = v; savePrefs(); draw(); } };
  $('#bgs').onclick = async (e) => {
    if (e.target.dataset.del !== undefined) {
      const i = Number(e.target.dataset.del);
      S.uploads.splice(i, 1);
      S.assets.uploads.splice(i, 1);
      await chrome.storage.local.set({ cardUploads: S.uploads });
      if (S.prefs.bg.kind === 'upload') S.prefs.bg = { kind: 'builtin', id: 'starfield' };
      savePrefs(); renderThumbs(); draw();
      return;
    }
    const t = e.target.closest('.thumb');
    if (!t) return;
    S.prefs.bg = t.dataset.kind === 'builtin' ? { kind: 'builtin', id: t.dataset.id } : { kind: 'upload', index: Number(t.dataset.index) };
    savePrefs(); draw();
  };
  $('#upload').onchange = async (e) => {
    const file = e.target.files[0];
    e.target.value = '';
    if (!file) return;
    const dataUrl = await downscaleToDataUrl(file, 1600, 1350, 'image/jpeg', 0.88);
    if (dataUrl.length > MAX_UPLOAD_CHARS) return status('That image is too big even after shrinking (2 MB max).');
    S.uploads.push(dataUrl);
    S.assets.uploads.push(await safeImage(dataUrl));
    while (S.uploads.length > MAX_UPLOADS) { S.uploads.shift(); S.assets.uploads.shift(); }
    await chrome.storage.local.set({ cardUploads: S.uploads });
    S.prefs.bg = { kind: 'upload', index: S.uploads.length - 1 };
    savePrefs(); renderThumbs(); draw();
  };
  const blob = () => new Promise((r) => $('#cv').toBlob(r, 'image/png'));
  async function copy() {
    try { await navigator.clipboard.write([new ClipboardItem({ 'image/png': blob() })]); return true; }
    catch (err) { console.warn('[SolSim] copy', err); return false; }
  }
  $('#download').onclick = async () => {
    const a = document.createElement('a');
    a.href = URL.createObjectURL(await blob());
    a.download = `solsim-${S.position.symbol.replace(/[^A-Za-z0-9_-]/g, '') || 'coin'}-${new Date().toISOString().slice(0, 10)}.png`;
    a.click();
    setTimeout(() => URL.revokeObjectURL(a.href), 5000);
  };
  $('#copy').onclick = async () => status((await copy()) ? 'Copied image' : 'Could not copy; use Download');
  $('#share').onclick = async () => {
    const ok = await copy();
    window.open('https://x.com/intent/post?text=' + encodeURIComponent(shareText(S.model)), '_blank');
    status(ok ? 'Image copied. Paste it into your post.' : 'Could not copy; use Download');
  };
}

init().catch((e) => { console.error('[SolSim]', e); $('#msg').textContent = 'Something went wrong loading this card.'; });
```
